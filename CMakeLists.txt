cmake_minimum_required(VERSION 3.16)
project(mp4_to_mp3 LANGUAGES CXX)
add_executable(mp4_to_mp3 WIN32 src/mp4_to_mp3.cpp)
target_compile_features(mp4_to_mp3 PRIVATE cxx_std_17)
target_compile_definitions(mp4_to_mp3 PRIVATE UNICODE _UNICODE)
target_link_libraries(mp4_to_mp3 PRIVATE comdlg32 shell32 gdi32)
if(MINGW)
  target_link_options(mp4_to_mp3 PRIVATE -municode)
endif()
if(MSVC)
  target_compile_options(mp4_to_mp3 PRIVATE /utf-8)
endif()
